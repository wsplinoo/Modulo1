double nota1 = 7.6;
double nota2 = 8.0;
double nota3 = 9.5;
double media = (nota1 + nota2 + nota3) / 3;
int frequencia = 90;
bool aprovacao;

if (media >= 7.0 && frequencia >= 75)
{
    aprovacao = true;
}
else
{
    aprovacao = false;
}

Console.WriteLine($"Nota 1: {nota1:F2}");
Console.WriteLine($"Nota 2: {nota2:F2}");
Console.WriteLine($"Nota 3: {nota3:F2} \n");
Console.WriteLine($"Média do aluno: {media:F2}");
Console.WriteLine($"Aprovado? {aprovacao}");
